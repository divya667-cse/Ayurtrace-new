// SPDX-License-Identifier: MIT pragma solidity ^0.8.0; contract HerbTraceability { struct Herb { uint id; string name; string location; uint quantity; string status; } mapping(uint => Herb) public herbs; uint public herbCount; function addHerb( string memory _name, string memory _location, uint _quantity ) public { herbCount++; herbs[herbCount] = Herb( herbCount, _name, _location, _quantity, "Collected by Farmer" ); } function processHerb(uint _id) public { herbs[_id].status = "Processed by Processor"; } function manufactureHerb(uint _id) public { herbs[_id].status = "Packed by Manufacturer"; } function getHerb(uint _id) public view returns (Herb memory) { return herbs[_id]; } }
